4. Practical Code Example (Control Structures & Methods in C# / VB.NET)

Neeche ek code snippet diya gaya hai jo conditional statements aur methods ke use ko batata hai[span_7](start_span)[span_7](end_span):

// C# Example Demonstrating Control Structures and Methods
using System;

namespace ControlStructuresDemo
{
    class Program
    {
        // Sub Procedure / Void Method equivalent
        static void DisplayWelcomeMessage(string name)
        {
            Console.WriteLine("Welcome, " + name + " to .NET Programming!");
        }

        // Function returning a value
        static int CalculateSquare(int number)
        {
            return number * number;
        }

        static void Main(string[] args)
        {
            Console.WriteLine("--- Control Structures & Methods Demo ---");

            DisplayWelcomeMessage("Priya");

            // Control Structure: If-Else Condition
            int score = 85;
            if (score >= 75)
            {
                Console.WriteLine("Grade: Distinction (Passed successfully)");
            }
            else
            {
                Console.WriteLine("Grade: Standard Pass");
            }

            // Control Structure: For Loop
            Console.WriteLine("Loop Execution (1 to 3):");
            for (int i = 1; i <= 3; i++)
            {
                int square = CalculateSquare(i);
                Console.WriteLine("Square of " + i + " is " + square);
            }

            Console.WriteLine("Control structures and methods executed successfully.");
            Console.ReadKey();
        }
    }
}
